python3 - <<'PY'
from collections import Counter, defaultdict
from datetime import datetime
import os
import subprocess

now = datetime.now().astimezone()
print("FORECAST_SNAPSHOT", now.isoformat())
print("Times use this login host's timezone, with UTC offset shown for each hour.")
command = ["squeue", "--me", "--array", "--states=PENDING", "--start", "--noheader",
           "--format=%i|%j|%T|%S|%R"]
try:
    result = subprocess.run(command, text=True, capture_output=True, check=True, timeout=30,
                            env={**os.environ, "SLURM_TIME_FORMAT": "%Y-%m-%dT%H:%M:%S"})
    if result.stderr.strip():
        print("SLURM_WARNING", result.stderr.strip())
    rows = [[part.strip() for part in line.split("|")]
            for line in result.stdout.splitlines() if line.strip()]
    if any(len(row) != 5 or row[2] != "PENDING" for row in rows):
        raise ValueError("unexpected Slurm output; no forecast reported")
    rows = list({row[0]: row for row in rows}.values())
    hours = defaultdict(list)
    unknown, stale = [], []
    for row in rows:
        try:
            start = datetime.fromisoformat(row[3]).astimezone()
        except ValueError:
            unknown.append(row)
            continue
        if start < now:
            stale.append(row)
        else:
            hours[start.replace(minute=0, second=0, microsecond=0)].append((start, row))
    print("PENDING_TOTAL", len(rows), "WITH_FUTURE_ESTIMATE", sum(map(len, hours.values())),
          "NO_ESTIMATE", len(unknown), "STALE_ESTIMATE", len(stale))
    print("HOUR_START | ALL_MY_JOBS | GEMMA_V4")
    for hour, entries in sorted(hours.items()):
        gemma = sum(row[1] == "geodml-gemma-si-v4" for _, row in entries)
        print(hour.isoformat(timespec="minutes"), "|", len(entries), "|", gemma)
        for start, row in sorted(entries):
            print("  ", start.isoformat(timespec="minutes"), row[0], row[1], row[4])
    for label, entries in (("NO_ESTIMATE", unknown), ("STALE_ESTIMATE", stale)):
        if entries:
            print(label, "reasons:", dict(Counter(row[4] for row in entries)))
            for row in entries:
                print("  ", " | ".join(row))
    if not rows:
        print("NO_PENDING_JOBS for your user in this snapshot.")
    elif not hours:
        print("NO_FUTURE_START_ESTIMATES: Slurm has not supplied a usable schedule.")
    print("Only hours with estimates are listed. An omitted hour is not a guarantee of no starts.")
    print("Estimates can change with priority, dependencies and resource availability; rerun for an update.")
except (OSError, subprocess.SubprocessError, ValueError) as error:
    print("FORECAST_UNAVAILABLE:", str(error))
    if isinstance(error, subprocess.CalledProcessError):
        print(error.stderr.strip())
