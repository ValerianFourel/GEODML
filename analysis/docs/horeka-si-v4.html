from collections import Counter
from contextlib import closing
from datetime import datetime, timezone
import json
from pathlib import Path
import sqlite3
import sys

root = Path(sys.argv[1])
print("SNAPSHOT", datetime.now(timezone.utc).isoformat(), root)

    raise SystemExit("RUN_UNAVAILABLE: config.json missing; check the run path.")

         ("v4-pass3", "v4-inputs")]

            continue

            with closing(sqlite3.connect(path.resolve().as_uri() + "?mode=ro", uri=True, timeout=2)) as db:
                db.execute("PRAGMA query_only=ON")
                rows = db.execute("SELECT state,result FROM tasks").fetchall()

                if state in ("done", "saved"):

                               "failed" if result.get("ok") is False else "unknown")
                    label = ({"ok": "completed", "failed": "failed", "unknown": "unknown"}[outcome]

            fields = ("completed", "failed", "running", "pending", "waiting", "blocked",
                      "saved_ok", "saved_failed", "saved_unknown", "unknown")

                  " ".join(key + "=" + str(counts[key]) for key in fields))
            if len(rows) != expected:
                print("  INVENTORY_MISMATCH: do not interpret missing rows as completed.")
        except (OSError, sqlite3.Error, ValueError, AttributeError) as error:
            print(name, "UNAVAILABLE:", type(error).__name__, str(error))

completed and failed split sealed results by inference outcome. saved_ok and saved_failed are written results awaiting sealing, shown separately. pending is queued work; waiting awaits an answer map; blocked means a source cannot be scored because its map failed, was quarantined or declared the input ineligible. running is the recorded task state and can remain after interruption. Check Slurm and logs below for current activity. Unknown or unavailable data is not counted as success. These counts do not establish semantic acceptance.

from collections import Counter
from datetime import datetime, timezone
import getpass
import subprocess

job_name = "geodml-gemma-si-v4"

Forecast new job starts hour by hour

Separate HoreKa login shell · read-only

Shows Slurm’s estimated starts for all your pending HoreKa jobs, grouped by hour, with a separate Gemma v4 count. Each hour lists the job IDs, names and estimated start times. Times use the HoreKa login host’s timezone, not your Mac’s timezone.

python3 - <<'PY'
from collections import Counter, defaultdict
from datetime import datetime
import os
import subprocess

now = datetime.now().astimezone()
print("FORECAST_SNAPSHOT", now.isoformat())
print("Times use this login host's timezone, with UTC offset shown for each hour.")
command = ["squeue", "--me", "--array", "--states=PENDING", "--start", "--noheader",
           "--format=%i|%j|%T|%S|%R"]
try:
    result = subprocess.run(command, text=True, capture_output=True, check=True, timeout=30,
                            env={**os.environ, "SLURM_TIME_FORMAT": "%Y-%m-%dT%H:%M:%S"})
    if result.stderr.strip():
        print("SLURM_WARNING", result.stderr.strip())
    rows = [[part.strip() for part in line.split("|")]
            for line in result.stdout.splitlines() if line.strip()]
    if any(len(row) != 5 or row[2] != "PENDING" for row in rows):
        raise ValueError("unexpected Slurm output; no forecast reported")
    rows = list({row[0]: row for row in rows}.values())
    hours = defaultdict(list)
    unknown, stale = [], []
    for row in rows:
        try:
            start = datetime.fromisoformat(row[3]).astimezone()
        except ValueError:
            unknown.append(row)
            continue
        if start < now:
            stale.append(row)
        else:
            hours[start.replace(minute=0, second=0, microsecond=0)].append((start, row))
    print("PENDING_TOTAL", len(rows), "WITH_FUTURE_ESTIMATE", sum(map(len, hours.values())),
          "NO_ESTIMATE", len(unknown), "STALE_ESTIMATE", len(stale))
    print("HOUR_START | ALL_MY_JOBS | GEMMA_V4")
    for hour, entries in sorted(hours.items()):
        gemma = sum(row[1] == "geodml-gemma-si-v4" for _, row in entries)
        print(hour.isoformat(timespec="minutes"), "|", len(entries), "|", gemma)
        for start, row in sorted(entries):
            print("  ", start.isoformat(timespec="minutes"), row[0], row[1], row[4])
    for label, entries in (("NO_ESTIMATE", unknown), ("STALE_ESTIMATE", stale)):
        if entries:
            print(label, "reasons:", dict(Counter(row[4] for row in entries)))
            for row in entries:
                print("  ", " | ".join(row))
    if not rows:
        print("NO_PENDING_JOBS for your user in this snapshot.")
    elif not hours:
        print("NO_FUTURE_START_ESTIMATES: Slurm has not supplied a usable schedule.")
    print("Only hours with estimates are listed. An omitted hour is not a guarantee of no starts.")
    print("Estimates can change with priority, dependencies and resource availability; rerun for an update.")
except (OSError, subprocess.SubprocessError, ValueError) as error:
    print("FORECAST_UNAVAILABLE:", str(error))
    if isinstance(error, subprocess.CalledProcessError):
        print(error.stderr.strip())
PY

These are scheduler estimates, not reservations or promised start times. NO_ESTIMATE means Slurm has not assigned a start estimate. STALE_ESTIMATE means the reported start is already in the past while the job remains pending. Both are shown separately with their reasons. Only hours with future estimates appear in the table; jobs without estimates can still start in other hours. Repeat manually to refresh. This check submits or changes no jobs.
