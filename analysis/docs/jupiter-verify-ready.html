import json, os, subprocess
from pathlib import Path
roots = [Path('/e')/storage/'scifi'/os.environ['USER'] for storage in ('fscratch','scratch','project1')]
repos = set()
for root in roots:
    repos.update(p for p in (root/'geodml/src').glob('*') if (p/'.git').exists())
    if (root/'GEODML_Analysis/.git').exists(): repos.add(root/'GEODML_Analysis')
rows=[]
for repo in sorted(repos):
    def git(*args):
        return subprocess.check_output(['git','-C',str(repo),*args],text=True,stderr=subprocess.PIPE,timeout=120).strip()
    row={'path':str(repo)}
    try:
        git('fetch','--quiet','--prune','origin')
        row.update(head=git('rev-parse','HEAD'), dirty=git('status','--porcelain','--untracked-files=all').splitlines(),
                   unpushed_commits=int(git('rev-list','--count','HEAD','--branches','--not','--remotes=origin')))
        row['status']='clean_and_pushed' if not row['dirty'] and row['unpushed_commits']==0 else 'attention'
    except Exception as exc:
        row.update(status='unknown',error=type(exc).__name__)
    print('CODE_STATE '+json.dumps(row),flush=True); rows.append(row)
Path(os.environ['VERIFY_ROOT'],'code-state.json').write_text(json.dumps(rows,indent=2))
print('CODE_SCAN_CHECKOUTS',len(rows))
for command in ('crontab','scrontab'):
    try:
        p=subprocess.run([command,'-l'],capture_output=True,text=True,timeout=20)
        entries=[s for s in p.stdout.splitlines() if s.strip() and not s.lstrip().startswith('#')]
        print('SCHEDULED_ENTRIES',command,len(entries),'exit',p.returncode)
    except Exception as exc: print('SCHEDULED_ENTRIES',command,'unknown',type(exc).__name__)
PY_LOCAL
) 2>&1 | tee "/e/project1/scifi/$USER/geodml/jupiter-verification/local-status-$(date +%Y%m%d-%H%M%S).log"
